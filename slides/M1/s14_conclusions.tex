\begin{frame}{Conclusions}
\framesubtitle{Slide 14 \textbullet\ M1 \textbullet\ 1:00 \textbullet\ the signal is real; choosing the pairs is the weak link}
\vspace{-2pt}
\footnotesize
\begin{columns}[T,onlytextwidth]
\column{0.31\textwidth}
  \textbf{\color{deckAccent}1\ \ Detect}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item a year of data finds only half-lives up to about 10 days (E2a)
    \item Engle--Granger passes about as many pairs as chance (8 of 196)
    \item most selected pairs soon stop mean-reverting (E9)
  \end{itemize}
\column{0.31\textwidth}
  \textbf{\color{deckAccent}2\ \ Model}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item the OU model and Bertram's formula are correct (E3)
    \item but $\kappa$ is hard to estimate: 18\% too high even for a fast pair (E1)
    \item a perfect pair earns only about half its theory (E4)
  \end{itemize}
\column{0.31\textwidth}
  \textbf{\color{deckAccent}3\ \ Trade}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item before costs the rules catch real mean reversion (about two trades in three win)
    \item after costs no rule is significant, in the US or India (E5, E8)
    \item the weak link is \textbf{pair selection}, not the trading rules
  \end{itemize}
\end{columns}
\vspace{4pt}
\scriptsize\textbf{\color{deckAccent}Limits}\; survivorship bias, a fixed hedge ratio within each window, daily closes only.\\
\textbf{\color{deckAccent}Next}\; a larger universe of stocks, a more stable pair selection, a time-varying hedge ratio.
\pairTakeaway{A real, trade-level signal that selection cannot yet turn into a significant profit: the \textbf{bottleneck is choosing genuinely cointegrated pairs}.}
\end{frame}
